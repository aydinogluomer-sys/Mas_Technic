/* QA ROUND 4 PROBE P6 — TEMPORARY.
   Was the OLD live control really blocked by the root fix? C5 says restoring
   `className="shell-stack"` under `.shell-stack > * { min-width: 0 }` yields
   ZERO problems, so the old control went green and the guard failed. Measure. */
import { test, type Page } from "@playwright/test";
import { mkdirSync, writeFileSync } from "node:fs";
import { gotoAndSettle } from "./helpers";

const OUT = "reports/qa/phase-08/r4";
mkdirSync(OUT, { recursive: true });

async function settle(page: Page, route: string) {
  await gotoAndSettle(page, route);
  await page.waitForLoadState("networkidle", { timeout: 10_000 }).catch(() => {});
  await page.waitForTimeout(350);
}

test("P6 old control under the root fix", async ({ page }) => {
  test.setTimeout(180_000);
  const read = () => page.evaluate(() => {
    const t = document.querySelector("main table")!;
    const w = t.closest("figure.shell-table")!.parentElement as HTMLElement;
    const scroll = t.closest(".shell-table-scroll") as HTMLElement;
    const before = scroll.scrollLeft;
    scroll.scrollLeft = 9999;
    const forced = scroll.scrollLeft;
    const rightAtMax = t.getBoundingClientRect().right;
    scroll.scrollLeft = before;
    const b = scroll.getBoundingClientRect();
    const tr = t.getBoundingClientRect();
    return {
      wrapperClass: w.className,
      vw: window.innerWidth,
      containerLeft: Number(b.left.toFixed(3)),
      containerRight: Number(b.right.toFixed(3)),
      clientWidth: scroll.clientWidth,
      scrollWidth: scroll.scrollWidth,
      forcedScrollLeft: forced,
      tableRight: Number(tr.right.toFixed(3)),
      rightAtMax: Number(rightAtMax.toFixed(3)),
      tabindex: scroll.getAttribute("tabindex"),
      // the checker's verdict, inlined
      r21: b.left < -1 || b.right > window.innerWidth + 1,
      fits: tr.left >= -1 && tr.right <= window.innerWidth + 1,
    };
  });
  const restore = () => page.evaluate(() => {
    const w = document.querySelector("main table")!.closest("figure.shell-table")!.parentElement as HTMLElement;
    w.className = "shell-stack";
    w.setAttribute("data-gap", "sm");
  });

  await page.setViewportSize({ width: 375, height: 812 });
  await settle(page, "/cerez-politikasi");
  const shipped = await read();
  await restore();
  await page.waitForTimeout(400);
  const oldControlNoFix = await read();

  await settle(page, "/cerez-politikasi");
  await page.addStyleTag({ content: ".shell-stack > * { min-width: 0 }" });
  await restore();
  await page.waitForTimeout(400);
  const oldControlWithFix = await read();

  const out = { shipped, oldControlNoFix, oldControlWithFix };
  writeFileSync(`${OUT}/p6-old-control.json`, JSON.stringify(out, null, 2), "utf8");
  console.log(JSON.stringify(out, null, 1));
});
